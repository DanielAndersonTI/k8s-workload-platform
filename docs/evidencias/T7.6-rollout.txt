# origem das requisicoes: a propria VM, porta publicada 8000/health
# data UTC: 2026-11-23

== estado antes
$ kubectl -n workloads rollout history deployment/cpu-workload
REVISION  CHANGE-CAUSE

3         <none>

$ kubectl -n workloads get deploy cpu-workload -o jsonpath="{.spec.template.spec.containers[0].image}"

# Um laco fez uma requisicao por segundo durante 80 s, comecando antes da
# troca de imagem.

$ kubectl -n workloads get pods -l app.kubernetes.io/name=cpu-workload
