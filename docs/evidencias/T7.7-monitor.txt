== retorno das replicas
$ kubectl -n workloads scale deployment/cpu-workload --replicas=2
deployment.apps/cpu-workload scaled

$ kubectl -n workloads rollout status deployment/cpu-workload --timeout=180s
deployment "cpu-workload" successfully rolled out

$ kubectl -n workloads get pods -l app.kubernetes.io/name=cpu-workload
NAME                           READY   STATUS    RESTARTS   AGE
cpu-workload-6dc4cd7db-klpct   1/1     Running   0          3s
cpu-workload-6dc4cd7db-w9nlq   1/1     Running   0          3s

# 20 s depois
$ openstack loadbalancer member list c336436d-7bea-473f-a92d-fd57024d02f2 -f value -c operating_status
ONLINE

$ curl -s -o /dev/null -w 'http=%{http_code}\n' http://150.165.85.77:8600/health
http=200

# O monitor usa HTTP /health, delay 5 s, timeout 4 s e 3 tentativas. A queda de
# um pod nao altera o membro, porque o Service ainda tem uma replica pronta.
